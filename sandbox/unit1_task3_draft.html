brigham = (
    df.filter(pl.col('name') == "Brigham")
        .with_columns(
            ((pl.col('ID') + pl.col('UT')) / pl.col('Total'))
            .alias('prop_UT_ID'))
        .select(['name', 'year', 'Total', 'prop_UT_ID'])
)
(
ggplot(data = brigham, mapping = aes(x = 'year', y = 'prop_UT_ID')) 
    + geom_line() 
    + scale_x_continuous(format = '.0f')
